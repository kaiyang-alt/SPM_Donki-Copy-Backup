-- =====================================================================
-- ConnectSphere — Event Assignment (automatic coordinator assignment)
--
-- REQUIRES TEAM APPROVAL BEFORE RUNNING.
-- Adds one SECURITY DEFINER trigger function + one trigger on event.
-- Does NOT add/rename tables, columns, relationships, status values,
-- or RLS policies.
--
-- Behaviour:
--   When an event becomes 'requested' with no coordinator, the system
--   assigns exactly one active Event Coordinator, moves the event to
--   'planning', and notifies the coordinator and the organiser.
--
-- Fairness rule (team-decided):
--   1. Lowest active workload (events in requested / changes_requested /
--      planning / confirmed assigned to that coordinator)
--   2. Tie: coordinator least recently given an event (never-assigned first)
--   3. Tie: user_id (deterministic)
-- =====================================================================

create or replace function public.auto_assign_event_coordinator()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  v_coordinator_id   uuid;
  v_coordinator_name text;
  v_event_name       text;
begin
  -- Serialise concurrent assignments so two simultaneous submissions
  -- don't both see the same "lowest workload" coordinator.
  perform pg_advisory_xact_lock(hashtext('event_coordinator_auto_assignment'));

  -- Pick the eligible coordinator using the fairness rule.
  select ur.user_id, p.name
    into v_coordinator_id, v_coordinator_name
  from public.user_role ur
  join public.profile p
    on p.user_id = ur.user_id
  left join public.event e
    on e.coordinator_id = ur.user_id
  where ur.role = 'event_coordinator'
    and p.active = true
  group by ur.user_id, p.name
  order by
    count(e.event_id) filter (
      where e.status in ('requested', 'changes_requested', 'planning', 'confirmed')
    ) asc,
    max(e.created_at) asc nulls first,
    ur.user_id asc
  limit 1;

  -- No active coordinator: leave the event 'requested' and unassigned,
  -- and do not block the organiser's submission.
  if v_coordinator_id is null then
    return null;
  end if;

  -- Assign only if still unassigned (enforces "one coordinator only").
  update public.event
     set coordinator_id = v_coordinator_id,
         status         = 'planning',
         updated_at     = now()
   where event_id       = new.event_id
     and coordinator_id is null
     and status         = 'requested';

  if not found then
    return null;  -- already assigned elsewhere; do nothing
  end if;

  v_event_name := coalesce(new.name, 'Untitled event');

  insert into public.notification
    (notification_id, event_id, recipient_id, type, message, created_at)
  values
    (gen_random_uuid(), new.event_id, v_coordinator_id,
     'coordinator_assigned',
     format('You have been assigned as the Event Coordinator for "%s".', v_event_name),
     now()),
    (gen_random_uuid(), new.event_id, new.organiser_id,
     'event_coordinator_assigned',
     format('Your event "%s" has been assigned to Event Coordinator %s.',
            v_event_name, coalesce(v_coordinator_name, 'an Event Coordinator')),
     now());

  return null;  -- AFTER trigger: return value is ignored
end;
$$;

-- Nobody should call this directly; it only runs via the trigger.
revoke all on function public.auto_assign_event_coordinator()
  from public, anon, authenticated;

drop trigger if exists trg_event_auto_assign_coordinator on public.event;

create trigger trg_event_auto_assign_coordinator
after insert or update of status on public.event
for each row
when (new.status = 'requested' and new.coordinator_id is null)
execute function public.auto_assign_event_coordinator();
